nexusagent

NexusAgent — a minimal, dependency-free agent runtime.

This module exposes the entire stable public API (1.0 stability contract, ADR 0010). Every name listed in __all__ is governed by semantic versioning: within a major release nothing here is renamed, moved, or removed, and new capabilities are only added. Everything else in nexusagent.* is internal and may change in any release.

Quick start: build a runtime with create_application_runtime() and pass prompts to the returned Runtime — see the README.

The import name is nexusagent (the distribution is nexusagent-runtime on PyPI).

 84    def run(self, input_text: str, *, on_delta: Callable[[str], None] | None = None) -> AgentResult:
 85        """Run the agent on the given input and return an AgentResult.
 86
 87        If the Provider requests tools and this Agent has a ToolRegistry and a
 88        ToolExecutor, the requested tools are executed and their results are
 89        fed back to the Provider until it returns a plain text answer or
 90        ``max_iterations`` provider calls have been made.
 91
 92        When ``on_delta`` is given, the provider's ``stream_response()`` is
 93        used and ``on_delta`` is called with every streamed text fragment as
 94        it arrives — including text the model emits before requesting a tool
 95        — while ``AgentResult.output`` still carries only the final answer.
 96        Callbacks fire only when the backend actually streams; providers
 97        without incremental output simply never call ``on_delta``. An
 98        exception raised by ``on_delta`` aborts the run and is reported like
 99        a provider error. Without ``on_delta`` the non-streaming
100        ``generate_response()`` path is used, unchanged.
101
102        With memory configured (ADR 0012) the remembered turns reach the
103        provider either composed into the prompt text (providers that have
104        not opted into native history — the behavior is byte-for-byte the
105        pre-1.5 one) or as the structured ``history`` keyword (ADR 0014)
106        for providers whose ``supports_native_history`` flag is set, in
107        which case ``prompt`` carries the raw user input. A successful run
108        records ``(user, input_text)`` and ``(assistant, output)``. Failed
109        runs — provider errors, missing tools, iteration exhaustion —
110        record nothing, so memory only ever holds completed exchanges.
111        """
112        if not isinstance(input_text, str):
113            raise TypeError("input_text must be a string")
114
115        if not input_text.strip():
116            raise ValueError("input_text must not be empty or whitespace-only")
117
118        can_use_tools = self.tool_registry is not None and self.tool_executor is not None
119        tools = self._tool_specs() if can_use_tools else ()
120        tool_results: list[ToolResult] = []
121        turns = self.memory.history() if self.memory is not None else ()
122        if turns and getattr(self.provider, "supports_native_history", False):
123            # Native history threading (ADR 0014): history-aware providers
124            # rehydrate the remembered turns into their native multi-message
125            # wire format; the raw input travels as the final user message.
126            history: tuple[tuple[str, str], ...] = tuple((t.role, t.text) for t in turns)
127            prompt = input_text
128        else:
129            history = ()
130            prompt = compose_prompt(input_text, turns)
131
132        for iteration in range(1, self.max_iterations + 1):
133            try:
134                if on_delta is None:
135                    response = self._ask_provider(prompt, tools, tuple(tool_results), history)
136                else:
137                    response = self._stream_provider_response(
138                        prompt, tools, tuple(tool_results), on_delta, history
139                    )
140            except Exception as exc:  # noqa: BLE001
141                return AgentResult(output=f"provider error: {exc}", success=False)
142
143            if isinstance(response, TextResponse):
144                _logger.debug("iteration %d: provider answered with text", iteration)
145                self._remember(input_text, response.text)
146                return AgentResult(output=response.text, success=True)
147
148            if not isinstance(response, ToolCallRequest):
149                return AgentResult(
150                    output=f"provider error: unexpected response type {type(response).__name__}",
151                    success=False,
152                )
153
154            if not can_use_tools:
155                return AgentResult(
156                    output=(
157                        f"provider requested tool {response.tool_name!r} "
158                        "but the agent has no tools configured"
159                    ),
160                    success=False,
161                )
162
163            _logger.info("iteration %d: provider requested tool %r", iteration, response.tool_name)
164            tool_results.append(self._run_tool_call(response))
165
166        return AgentResult(
167            output=f"agent stopped: exceeded the maximum of {self.max_iterations} iterations",
168            success=False,
169        )

189    def _ask_provider(
190        self,
191        input_text: str,
192        tools: tuple[ToolSpec, ...],
193        tool_results: tuple[ToolResult, ...],
194        history: tuple[tuple[str, str], ...] = (),
195    ) -> ProviderResponse:
196        """Ask the provider for its next response.
197
198        Any object with a ``generate(prompt) -> str`` method is accepted as a
199        provider; only ones that also define ``generate_response`` can request
200        tools. The ``history`` keyword (ADR 0014) is forwarded only to
201        providers that opted into native history threading via
202        ``supports_native_history`` — every other provider keeps receiving
203        exactly the pre-1.5 keyword set.
204        """
205        generate_response = getattr(self.provider, "generate_response", None)
206        if generate_response is None:
207            return TextResponse(self.provider.generate(input_text))
208
209        if getattr(self.provider, "supports_native_history", False):
210            response: ProviderResponse = generate_response(
211                input_text, tools=tools, tool_results=tool_results, history=history
212            )
213            return response
214
215        response = generate_response(input_text, tools=tools, tool_results=tool_results)
216        return response

218    def _stream_provider_response(
219        self,
220        input_text: str,
221        tools: tuple[ToolSpec, ...],
222        tool_results: tuple[ToolResult, ...],
223        on_delta: Callable[[str], None],
224        history: tuple[tuple[str, str], ...] = (),
225    ) -> ProviderResponse:
226        """Collect one streamed provider response, forwarding text deltas.
227
228        Falls back to the plain ``generate_response()`` path for providers
229        that do not implement ``stream_response()`` at all; ``on_delta``
230        then stays silent. ``history`` (ADR 0014) is forwarded only to
231        providers that opted into native history threading. The stream's
232        final event is the iteration's response; a stream that ends without
233        one is an error.
234        """
235        stream = getattr(self.provider, "stream_response", None)
236        if stream is None:
237            return self._ask_provider(input_text, tools, tool_results, history)
238
239        if getattr(self.provider, "supports_native_history", False):
240            events = stream(input_text, tools=tools, tool_results=tool_results, history=history)
241        else:
242            events = stream(input_text, tools=tools, tool_results=tool_results)
243
244        for event in events:
245            if isinstance(event, StreamTextDelta):
246                on_delta(event.text)
247            elif isinstance(event, (TextResponse, ToolCallRequest)):
248                return event
249
250        raise ProviderResponseError("provider stream ended without a final response")

253        """Record one completed exchange in memory, when configured (ADR 0012)."""
254        if self.memory is None:
255            return
256
257        self.memory.append(ConversationTurn(USER_ROLE, input_text))
258        self.memory.append(ConversationTurn(ASSISTANT_ROLE, output))
259        _logger.debug("memory now holds %d turns", len(self.memory))

262        assert self.tool_registry is not None
263        return tuple(ToolSpec(t.name, t.description) for t in self.tool_registry.list_tools())

265    def _run_tool_call(self, request: ToolCallRequest) -> ToolResult:
266        """Run one requested tool. Failures become error results, never crashes.
267
268        The Provider (usually a model) is told what went wrong so it can retry
269        with better input, pick another tool, or explain the problem.
270        """
271        try:
272            output = self.execute_tool(request.tool_name, request.tool_input)
273        except KeyError:
274            return self._error_result(request, f"Tool {request.tool_name!r} is not registered")
275        except (TypeError, ValueError, ToolExecutionError) as exc:
276            return self._error_result(request, str(exc))
277
278        _logger.debug("tool %r succeeded", request.tool_name)
279        return ToolResult(request.tool_name, request.tool_input, output)

 69    """Route each prompt to one named agent and return that agent's result.
 70
 71    ``agents`` is a non-empty mapping of non-empty string names to
 72    :class:`~nexusagent.agent.Agent` instances; ``router`` is a callable
 73    that receives the raw user input and returns one of those names. Both
 74    are validated eagerly at construction, like
 75    :meth:`nexusagent.tool_registry.ToolRegistry.register`.
 76
 77    The team owns no behavior of its own. The selected agent's bounded
 78    tool loop, tools, memory and streaming all run unchanged, so a member
 79    behaves inside the team exactly as it would standalone (ADR 0015).
 80    """

 82    def __init__(self, agents: Mapping[str, Agent], router: Router) -> None:
 83        if not isinstance(agents, Mapping):
 84            raise TypeError("agents must be a mapping of name -> Agent")
 85
 86        if not agents:
 87            raise ValueError("agents must not be empty")
 88
 89        for name, agent in agents.items():
 90            if not isinstance(name, str):
 91                raise TypeError("agent names must be strings")
 92
 93            if not name.strip():
 94                raise ValueError("agent names must not be empty or whitespace-only")
 95
 96            if not isinstance(agent, Agent):
 97                raise TypeError(f"agent {name!r} must be an Agent instance")
 98
 99        if not callable(router):
100            raise TypeError("router must be callable")
101
102        self._agents: Mapping[str, Agent] = MappingProxyType(dict(agents))
103        self._router = router

156def create_provider_router(
157    provider: Provider,
158    agent_names: Sequence[str],
159    *,
160    instruction: str = DEFAULT_ROUTER_INSTRUCTION,
161) -> Router:
162    """Build a router callable from any Provider.
163
164    The returned callable sends one ``generate()`` request that lists the
165    agent names and the user prompt, then parses the reply with a
166    documented, deterministic rule: an exact match after stripping, then a
167    case-insensitive exact match, otherwise the listed name that appears
168    earliest in the reply (ties resolve to the earlier listed name). If no
169    name appears at all, ``ProviderResponseError`` is raised — the team
170    turns that into a failed ``TeamResult``. ``provider.generate()`` may
171    also raise its own provider errors; both paths surface as routing
172    failures instead of crashing the caller.
173    """
174    if not isinstance(provider, Provider):
175        raise TypeError("provider must be a Provider instance")
176
177    if not isinstance(agent_names, Sequence) or isinstance(agent_names, (str, bytes)):
178        raise TypeError("agent_names must be a sequence of strings")
179
180    if not agent_names:
181        raise ValueError("agent_names must not be empty")
182
183    for name in agent_names:
184        if not isinstance(name, str):
185            raise TypeError("agent names must be strings")
186
187        if not name.strip():
188            raise ValueError("agent names must not be empty or whitespace-only")
189
190    if len(set(agent_names)) != len(agent_names):
191        raise ValueError("agent_names must not contain duplicates")
192
193    listing = "\n".join(f"- {name}" for name in agent_names)
194
195    def router(prompt: str) -> str:
196        if not isinstance(prompt, str):
197            raise TypeError("prompt must be a string")
198
199        if not prompt.strip():
200            raise ValueError("prompt must not be empty or whitespace-only")
201
202        composed = (
203            f"{instruction}\n\nAvailable agents:\n{listing}\n\n"
204            f"User prompt:\n{prompt}\n\n"
205            "Answer with exactly one agent name and nothing else."
206        )
207        reply = provider.generate(composed)
208
209        if not isinstance(reply, str):
210            raise ProviderResponseError("router provider returned a non-string reply")
211
212        reply = reply.strip()
213        lowered = reply.lower()
214        for name in agent_names:
215            if lowered == name.lower():
216                return name
217
218        best_name, best_position = "", len(reply)
219        for name in agent_names:
220            position = lowered.find(name.lower())
221            if position != -1 and position < best_position:
222                best_name, best_position = name, position
223
224        if not best_name:
225            raise ProviderResponseError(
226                f"router provider did not name any of the agents {sorted(agent_names)!r}; "
227                f"reply was {reply!r}"
228            )
229
230        return best_name
231
232    return router
